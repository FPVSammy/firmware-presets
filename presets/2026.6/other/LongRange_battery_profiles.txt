# BATTERY PROFILE 0: LiIon

set battery_profile_name = LiIon
set vbat_max_cell_voltage = 425
set vbat_min_cell_voltage = 300
set vbat_warning_cell_voltage = 325
set force_battery_cell_count = 6

# ==========================================
# BATTERY PROFILE 1: LiIonLow
# ==========================================

set vbat_max_cell_voltage = 425
set vbat_min_cell_voltage = 275
set vbat_warning_cell_voltage = 300

# ==========================================
# BATTERY PROFILE 2: LiPo
# ==========================================
battery_profile 2
